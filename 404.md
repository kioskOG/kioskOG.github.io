<script>
  (function () {
    'use strict';

    // Base fileSystem with default directories and virtual metrics files
    let fileSystem = {
      '/': {
        type: 'dir',
        children: {
          'sys': { type: 'dir' }
        }
      },
      '/sys': {
        type: 'dir',
        children: {
          'metrics.txt': { type: 'file', content: '--- CLUSTER STATUS ---\nKubernetes API Server: Active (v1.28.2)\nNode Status: 5/5 Nodes Ready\nPod Health: 99.8% Green\nActive MLOps Pipelines: Training-Recall-Weekly (Completed)\nSystem Load: CPU 12.4%, RAM 18.2 GB / 32.0 GB' },
          'build.log': { type: 'file', content: '--- BUILD STATISTICS ---\nBuild ID: 2026.06.15.14.04\nCompiler: Jekyll v4.3.2\nWebpack Output: assets/portfolio.css (27.9 KB)\nCSS Linters: Passed\nAutomated Tests: 42/42 Tests Passed (100% Success)' },
          'secrets.txt': { type: 'file', content: '--- HIGHLY CLASSIFIED ---\n[DEVOPS NOTE]: Found a stray config in staging. Replaced it.\n[MLOPS SECRETS]: KioskOG core model token: ks_live_9a4f21b7c0...\n[EASTER EGG]: "There are 10 types of people in the world: those who understand binary, and those who don\'t."' }
        }
      }
    };

    // Load VFS flat pages array dynamically and build directory tree structure
    fetch('{{ "/assets/vfs.json" | relative_url }}')
      .then(function (response) {
        return response.json();
      })
      .then(function (pages) {
        pages.forEach(function (page) {
          if (!page.path) return;
          var segments = page.path.split('/');
          var current = '/';
          
          for (var i = 0; i < segments.length; i++) {
            var seg = segments[i];
            if (!seg) continue;
            var isLast = (i === segments.length - 1);
            
            // Ensure current directory node is initialized
            if (!fileSystem[current]) {
              fileSystem[current] = { type: 'dir', children: {} };
            }
            if (!fileSystem[current].children) {
              fileSystem[current].children = {};
            }
            
            if (isLast) {
              // Add file leaf node
              fileSystem[current].children[seg] = {
                type: 'file',
                url: page.url,
                desc: page.desc || page.title
              };
            } else {
              // Add directory pointer
              fileSystem[current].children[seg] = { type: 'dir' };
              
              // Move to next directory path level
              var nextPath = (current === '/' ? '' : current) + '/' + seg;
              current = nextPath;
            }
          }
        });
        updatePrompt();
      })
      .catch(function (err) {
        console.error('Error loading dynamic VFS:', err);
      });

    let currentPath = '/';
    const body = document.getElementById('terminal-body');
    const input = document.getElementById('terminal-input');
    const history = document.getElementById('terminal-history');
    const promptDir = document.getElementById('prompt-dir');

    // Auto-focus input
    document.addEventListener('click', function () {
      if (input) input.focus();
    });

    if (input) {
      input.addEventListener('keydown', function (e) {
        if (e.key === 'Enter') {
          const commandLine = input.value;
          input.value = '';
          processCommand(commandLine);
        }
      });
    }

    function updatePrompt() {
      if (promptDir) {
        const displayPath = currentPath === '/' ? '~' : currentPath;
        promptDir.textContent = `guest@kioskog-cli:${displayPath}$`;
      }
    }

    function appendLine(text, className = '') {
      const div = document.createElement('div');
      div.className = 'terminal-line ' + className;
      div.innerHTML = text;
      if (history) history.appendChild(div);
      if (body) body.scrollTop = body.scrollHeight;
    }

    function processCommand(rawInput) {
      const trimmed = rawInput.trim();
      const parts = trimmed.split(/\s+/);
      const command = parts[0].toLowerCase();
      const args = parts.slice(1);

      // Append prompt and entered command to history
      const currentPromptText = promptDir ? promptDir.textContent : 'guest@kioskog-cli:~$';
      appendLine(`<span class="prompt">${currentPromptText}</span> ${trimmed}`);

      if (command === '') {
        return;
      }

      if (command === 'clear') {
        if (history) history.innerHTML = '';
        return;
      }

      if (command === 'help') {
        appendLine(
          'Available commands:\n' +
          '  help           - Show this support panel\n' +
          '  ls             - List directory files/subfolders\n' +
          '  cd &lt;dir&gt;       - Change directories (e.g., cd docs, cd ..)\n' +
          '  cat &lt;file&gt;     - Output file text or open document link\n' +
          '  whoami         - Print system user profile info\n' +
          '  sysinfo        - Display system hardware & cluster info\n' +
          '  theme          - Toggle website theme (dark/light)\n' +
          '  clear          - Clear console interface\n' +
          '  home / exit    - Go back to site home'
        );
      } else if (command === 'ls') {
        const node = fileSystem[currentPath];
        if (node && node.children) {
          const list = Object.keys(node.children).map(name => {
            const childNode = node.children[name];
            return childNode.type === 'dir'
              ? `<span class="text-blue">${name}/</span>`
              : `<span class="text-green">${name}</span>`;
          }).join('    ');
          appendLine(list || '(empty)');
        } else {
          appendLine('ls: metadata corrupted', 'text-red');
        }
      } else if (command === 'cd') {
        const target = args[0];
        if (!target) {
          currentPath = '/';
          updatePrompt();
          return;
        }

        if (target === '..') {
          if (currentPath === '/') return;
          const segments = currentPath.split('/').filter(Boolean);
          segments.pop();
          currentPath = '/' + segments.join('/');
          updatePrompt();
          return;
        }

        const node = fileSystem[currentPath];
        if (node && node.children && node.children[target]) {
          if (node.children[target].type === 'dir') {
            currentPath = (currentPath === '/' ? '' : currentPath) + '/' + target;
            updatePrompt();
          } else {
            appendLine(`cd: not a directory: ${target}`, 'text-red');
          }
        } else {
          appendLine(`cd: no such file or directory: ${target}`, 'text-red');
        }
      } else if (command === 'cat') {
        const target = args[0];
        if (!target) {
          appendLine('Usage: cat &lt;filename&gt;', 'text-yellow');
          return;
        }

        const node = fileSystem[currentPath];
        if (node && node.children && node.children[target]) {
          const fileNode = node.children[target];
          if (fileNode.type === 'file') {
            if (fileNode.content) {
              appendLine(fileNode.content);
            } else if (fileNode.url) {
              appendLine(`[SYS] Opening document "${fileNode.desc}"...`, 'text-blue');
              setTimeout(function () {
                window.location.href = fileNode.url;
              }, 1000);
            }
          } else {
            appendLine(`cat: ${target}: Is a directory`, 'text-red');
          }
        } else {
          appendLine(`cat: ${target}: No such file or directory`, 'text-red');
        }
      } else if (command === 'whoami') {
        appendLine('Jatin Sharma\nDevOps Engineer & Platform Architect\nFocused on scalable MLOps, containerization, and GitOps automation.');
      } else if (command === 'sysinfo') {
        appendLine(
          '<span class="text-blue">--- SYSTEM METRICS (Virtual Server) ---</span>\n' +
          'OS: Alpine Linux 3.19.1 (Host: KioskOG)\n' +
          'Kernel: Linux 6.6.14-0-virt (aarch64)\n' +
          'Uptime: 247 days, 14 hours, 22 minutes\n' +
          'CPU: Virtual Intel(R) Xeon(R) Gold @ 2.80GHz (4 Cores)\n' +
          'RAM: 18.2 GB / 32.0 GB (56.8% Used)\n' +
          'Storage: 124.7 GB / 250 GB\n' +
          'Kubernetes API Server: Active (v1.28.2)\n' +
          'Containers: Docker Engine 25.0.3 (38 Active Containers)'
        );
      } else if (command === 'theme') {
        const toggleBtn = document.getElementById('theme-toggle');
        if (toggleBtn) {
          toggleBtn.click();
          const activeTheme = document.documentElement.getAttribute('data-theme') || 'light';
          appendLine(`[SYS] Theme toggled to ${activeTheme} mode.`, 'text-blue');
        } else {
          appendLine('theme: toggle control not found in layout', 'text-red');
        }
      } else if (command === 'exit' || command === 'home') {
        appendLine('[SYS] Redirecting to Mission Control...', 'text-blue');
        setTimeout(function () {
          window.location.href = '/';
        }, 600);
      } else {
        appendLine(`shell: command not found: ${command}. Type 'help' for support.`, 'text-red');
      }
    }
  })();
</script>
